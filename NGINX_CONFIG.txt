# Corrected Nginx Reverse Proxy Configuration for Admin UI
# Version: 2.0 - Added URL rewriting for static files and redirects

location /eodhd/admin/ {
    # Strip the /eodhd/admin prefix before forwarding to Flask
    rewrite ^/eodhd/admin/(.*) /$1 break;

    proxy_pass http://172.28.0.200:5000;
    proxy_http_version 1.1;

    # Headers
    proxy_set_header Host $host;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Script-Name /eodhd/admin;

    # Rewrite redirects from Flask to include the prefix
    proxy_redirect / /eodhd/admin/;
    proxy_redirect http:// https://;

    # Rewrite URLs in HTML/CSS/JS to include the prefix
    # This fixes static file paths like /static/css/admin.css -> /eodhd/admin/static/css/admin.css
    sub_filter 'href="/' 'href="/eodhd/admin/';
    sub_filter 'src="/' 'src="/eodhd/admin/';
    sub_filter 'action="/' 'action="/eodhd/admin/';
    sub_filter 'url(/' 'url(/eodhd/admin/';
    sub_filter_once off;
    # Note: text/html is already included by default, only add CSS and JS
    sub_filter_types text/css application/javascript;

    # Buffering (off for better real-time performance)
    proxy_buffering off;
    proxy_request_buffering off;

    # Timeouts and size limits
    client_max_body_size 16M;
    proxy_read_timeout 600s;
    proxy_send_timeout 600s;
    proxy_connect_timeout 600s;
}

# Redirect without trailing slash to with trailing slash
location = /eodhd/admin {
    return 301 /eodhd/admin/;
}
